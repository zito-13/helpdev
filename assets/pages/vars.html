Requête Fetch API
async function getData(url) {
  try {
    const res = await fetch(url);
    if (!res.ok) throw new Error('Erreur');
    return await res.json();
  } catch (err) {
    console.error(err);
  }
}

Garder des données en mémoire locale.

Fonction Debounce
function debounce(func, delay = 300) {
  let timer;
  return (...args) => {
    clearTimeout(timer);
    timer = setTimeout(() => { 
      func.apply(this, args); 
    }, delay);
  };
}

Limite les appels (ex: scroll, input).
